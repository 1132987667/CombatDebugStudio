#### 改动 D — 问题 3（技能路径）：`selectAndExecuteSkill` 伤害日志双行格式

**位置**：`selectAndExecuteSkill` 方法内，命中瞬间循环 + 日志输出部分。

**D-1**：在命中瞬间循环中累加 `totalRawDamage`：

```typescript
// ── 修改前 ──
for (const entry of damageToken.getEntries()) {
  if (!entry.target.isAlive()) continue
  if (entry.damage > 0) {
    entry.target.takeDamage(entry.damage)
    this.emitDamageTakenEvent(
      source, entry.target, entry.damage, entry.rawDamage, isCrit, battleData,
    )
  }
  if (entry.heal > 0) {
    entry.target.heal(entry.heal)
  }
}
damageToken.clear()

// ── 修改后 ──
let totalRawDamage = 0                              // ★ 新增
for (const entry of damageToken.getEntries()) {
  if (!entry.target.isAlive()) continue
  if (entry.damage > 0) {
    entry.target.takeDamage(entry.damage)
    this.emitDamageTakenEvent(
      source, entry.target, entry.damage, entry.rawDamage, isCrit, battleData,
    )
    totalRawDamage += entry.rawDamage               // ★ 新增
  }
  if (entry.heal > 0) {
    entry.target.heal(entry.heal)
  }
}
damageToken.clear()
```

### 文件 2：`src/domain/battle/logs/renderers/RoundNarrativeRenderer.ts`

#### 改动 F — 问题 3（渲染器）：result 行格式

**位置**：渲染器中生成 action 块 `result` 行的逻辑（从 `meta.damage` / `meta.hpBefore` / `meta.hpAfter` 构建 `LogSegment[]` 的位置）。

```typescript
// ── 修改前（推断） ──
// result 行：`${meta.damage} 点物理伤害  ${meta.hpBefore} → ${meta.hpAfter}`

// ── 修改后 ──
// 从 meta 中读取新增的 entityName / entityFaction
const entityName = meta.entityName ?? ''
const entityFaction = meta.entityFaction as 'ally' | 'enemy' | undefined
const prefix = entityFaction === 'ally' ? '[友方]' : entityFaction === 'enemy' ? '[敌方]' : ''
const factionClass = entityFaction === 'ally' ? 'log-friendly' : 'log-hostile'

const resultSegments: LogSegment[] = []

// ★ 目标名 + "受到"
if (entityName) {
  resultSegments.push({
    text: `${prefix}${entityName}`,
    classStr: factionClass,
    kind: 'entity',
    faction: entityFaction,
  })
  resultSegments.push({ text: ' 受到 ' })
}

// ★ 最终伤害数值
resultSegments.push({
  text: `${meta.damage}`,
  classStr: 'log-damage',
})
resultSegments.push({ text: ' 点伤害' })   // ← 原 "点物理伤害" 改为 "点伤害"

// 气血变化（保持不变）
if (meta.hpBefore != null && meta.hpAfter != null) {
  resultSegments.push({ text: `  ${meta.hpBefore} → ${meta.hpAfter}` })
}
```
