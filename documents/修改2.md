好的，我们来详细展开 **“TriggerAction + 事件系统”** 的完整实现方案。这个方案的核心思想是：

- **属性修饰符**：只负责常驻、持续的数值调整（如“攻击力+20%”）。
- **触发行为**：通过战斗事件总线，在特定时机执行预设的行为（如“攻击时造成额外伤害”、“回合开始回复生命”）。
- **临时属性增益**：通过监听事件动态添加/移除短时 Buff，复用已有的属性修饰符系统。

下面我将给出完整的类型定义、事件总线实现、BuffSystem 集成以及使用示例。

### 一、类型定义扩展

首先在 `types/modifier-template.ts`（或新建 `types/buff.ts`）中定义触发相关的类型。

```typescript
// types/buff.ts (新增或扩展)

import type { ModifierTemplate } from './modifier-template'

// ========== 触发阶段枚举 ==========
export type TriggerPhase =
  | 'ON_BATTLE_START'      // 战斗开始时
  | 'ON_TURN_START'        // 回合开始时（每个参与者的回合开始）
  | 'ON_TURN_END'          // 回合结束时
  | 'ON_ATTACK_BEFORE'     // 攻击前（伤害计算前）
  | 'ON_ATTACK_HIT'        // 攻击命中时（伤害计算后、实际扣血前）
  | 'ON_ATTACK_AFTER'      // 攻击后（伤害结算完成）
  | 'ON_DAMAGE_TAKEN'      // 受到伤害后
  | 'ON_KILL'              // 击杀目标时
  | 'ON_DEATH'             // 自身死亡时
  | 'ON_HEAL_RECEIVED'     // 受到治疗时
  | 'ON_ENERGY_GAINED'     // 获得能量时
  // 可根据需要继续扩展

// ========== 触发行为定义 ==========
export interface TriggerAction {
  /** 触发的阶段 */
  phase: TriggerPhase
  /** 执行脚本ID（如 "deal_damage", "apply_buff", "heal"） */
  scriptId: string
  /** 执行参数（传递给脚本） */
  params?: Record<string, any>
  /** 触发概率（0-1），默认1 */
  probability?: number
  /** 冷却回合数（同一Buff实例的内部冷却） */
  cooldown?: number
  /** 最大触发次数（-1表示无限制） */
  maxTriggers?: number
}

// ========== 结构化Buff配置（更新） ==========
export interface StructuredBuffConfig {
  id: string
  name?: string
  /** 常驻属性修正 */
  modifiers?: ModifierTemplate[]
  /** 阶段触发行为 */
  triggers?: TriggerAction[]
  /** 持续时间（回合数），-1表示永久 */
  duration?: number
  /** 最大叠加层数 */
  maxStacks?: number
}
```

### 二、事件总线实现

实现一个轻量级的类型安全事件总线，用于战斗事件的分发。

```typescript
// core/EventBus.ts

import type { TriggerPhase } from '@/types/buff'

// ========== 事件上下文接口 ==========
export interface BattleEventContext {
  /** 事件发生时的战斗实例ID */
  battleId: string
  /** 事件相关的源参与者ID（如攻击者） */
  sourceId?: string
  /** 事件相关的目标参与者ID（如受击者） */
  targetId?: string
  /** 技能ID（如果事件由技能触发） */
  skillId?: string
  /** 伤害/治疗数值 */
  value?: number
  /** 其他自定义数据 */
  extra?: Record<string, any>
}

// ========== 事件监听器类型 ==========
export type EventListener = (context: BattleEventContext) => void

// ========== 事件总线实现 ==========
export class EventBus {
  private listeners: Map<TriggerPhase, Set<{ id: string; callback: EventListener }>> = new Map()
  private listenerIdCounter: number = 0

  /**
   * 注册事件监听器
   * @returns 监听器ID，用于取消注册
   */
  public on(phase: TriggerPhase, callback: EventListener): string {
    const id = `listener_${++this.listenerIdCounter}`
    if (!this.listeners.has(phase)) {
      this.listeners.set(phase, new Set())
    }
    this.listeners.get(phase)!.add({ id, callback })
    return id
  }

  /**
   * 取消注册
   */
  public off(phase: TriggerPhase, listenerId: string): boolean {
    const set = this.listeners.get(phase)
    if (!set) return false
    for (const item of set) {
      if (item.id === listenerId) {
        set.delete(item)
        return true
      }
    }
    return false
  }

  /**
   * 触发事件
   */
  public emit(phase: TriggerPhase, context: BattleEventContext): void {
    const set = this.listeners.get(phase)
    if (!set) return
    // 复制一份再遍历，防止回调中修改集合
    const listeners = Array.from(set)
    for (const { callback } of listeners) {
      try {
        callback(context)
      } catch (error) {
        console.error(`[EventBus] 事件 ${phase} 监听器执行出错:`, error)
      }
    }
  }

  /**
   * 清除所有监听器（用于战斗结束重置）
   */
  public clear(): void {
    this.listeners.clear()
  }
}
```

### 三、BuffSystem 集成事件注册

在 `BuffSystem` 中，当 Buff 被添加时，解析其 `triggers` 配置，向事件总线注册对应的回调。同时需要管理 Buff 实例的冷却、触发次数限制。

```typescript
// core/BuffSystem.ts (关键部分)

import { EventBus } from './EventBus'
import type { StructuredBuffConfig, TriggerAction } from '@/types/buff'
import type { IModifierStack } from '@/types/attribute'

/** Buff 实例运行时数据 */
interface BuffInstance {
  buffId: string
  config: StructuredBuffConfig
  targetId: string
  remainingDuration: number
  currentStacks: number
  // 触发冷却追踪
  cooldowns: Map<string, number>           // key: trigger index 或 scriptId
  triggerCounts: Map<string, number>       // 已触发次数
  // 注册的监听器ID列表，用于移除时清理
  listenerIds: string[]
}

export class BuffSystem implements IModifierProvider {
  private eventBus: EventBus
  private modifierStacks: Map<string, IModifierStack> = new Map()
  private activeBuffs: Map<string, BuffInstance[]> = new Map() // participantId -> instances

  constructor(eventBus: EventBus) {
    this.eventBus = eventBus
  }

  /**
   * 为目标添加Buff
   */
  addBuff(targetId: string, buffConfig: StructuredBuffConfig, stacks: number = 1): void {
    // 检查是否已存在同ID的Buff（叠加逻辑）
    const existing = this.getBuffInstances(targetId).find(b => b.buffId === buffConfig.id)
    if (existing && buffConfig.maxStacks && existing.currentStacks < buffConfig.maxStacks) {
      existing.currentStacks = Math.min(existing.currentStacks + stacks, buffConfig.maxStacks)
      // 叠加时可能需要刷新持续时间等，此处略
      return
    }

    // 创建新实例
    const instance: BuffInstance = {
      buffId: buffConfig.id,
      config: buffConfig,
      targetId,
      remainingDuration: buffConfig.duration ?? -1,
      currentStacks: stacks,
      cooldowns: new Map(),
      triggerCounts: new Map(),
      listenerIds: [],
    }

    // 1. 添加常驻属性修饰符
    const modifierStack = this.getModifierStack(targetId)
    if (modifierStack && buffConfig.modifiers) {
      for (const modTemplate of buffConfig.modifiers) {
        modifierStack.addModifier({
          buffInstanceId: `${buffConfig.id}_${targetId}_${Date.now()}`, // 应生成唯一ID
          attribute: modTemplate.targetAttribute,
          value: typeof modTemplate.value === 'number' ? modTemplate.value : 0, // 动态值需额外处理
          type: modTemplate.type,
        })
      }
    }

    // 2. 注册阶段触发器
    if (buffConfig.triggers) {
      for (let i = 0; i < buffConfig.triggers.length; i++) {
        const trigger = buffConfig.triggers[i]
        const listenerId = this.eventBus.on(trigger.phase, (context) => {
          this.handleTrigger(instance, trigger, context, i)
        })
        instance.listenerIds.push(listenerId)
      }
    }

    // 存储实例
    if (!this.activeBuffs.has(targetId)) {
      this.activeBuffs.set(targetId, [])
    }
    this.activeBuffs.get(targetId)!.push(instance)
  }

  /**
   * 处理触发行为
   */
  private handleTrigger(
    instance: BuffInstance,
    trigger: TriggerAction,
    context: BattleEventContext,
    triggerIndex: number,
  ): void {
    // 只处理目标相关的上下文
    if (context.sourceId !== instance.targetId && context.targetId !== instance.targetId) {
      return
    }

    const triggerKey = `trigger_${triggerIndex}`

    // 检查冷却
    const currentCooldown = instance.cooldowns.get(triggerKey) || 0
    if (currentCooldown > 0) {
      return
    }

    // 检查最大触发次数
    const triggerCount = instance.triggerCounts.get(triggerKey) || 0
    if (trigger.maxTriggers !== undefined && triggerCount >= trigger.maxTriggers) {
      return
    }

    // 概率判定
    if (trigger.probability !== undefined && trigger.probability < 1) {
      if (Math.random() > trigger.probability) {
        return
      }
    }

    // 执行脚本
    this.executeScript(trigger.scriptId, {
      ...context,
      buffInstance: instance,
      params: trigger.params,
    })

    // 更新冷却和计数
    if (trigger.cooldown) {
      instance.cooldowns.set(triggerKey, trigger.cooldown)
    }
    instance.triggerCounts.set(triggerKey, triggerCount + 1)
  }

  /**
   * 执行行为脚本（这里简化，实际应接入脚本引擎）
   */
  private executeScript(scriptId: string, context: any): void {
    // 示例：根据 scriptId 执行不同逻辑
    switch (scriptId) {
      case 'deal_damage':
        this.dealDirectDamage(context.targetId, context.params?.damage ?? 0)
        break
      case 'apply_buff':
        this.addBuff(context.targetId, context.params?.buffConfig)
        break
      case 'heal':
        this.healTarget(context.targetId, context.params?.amount ?? 0)
        break
      // ... 其他脚本
      default:
        console.warn(`[BuffSystem] 未知脚本ID: ${scriptId}`)
    }
  }

  /**
   * 回合结束时，减少Buff持续时间和冷却
   */
  onTurnEnd(participantId: string): void {
    const instances = this.activeBuffs.get(participantId) || []
    for (const inst of instances) {
      // 减少持续时间
      if (inst.remainingDuration > 0) {
        inst.remainingDuration--
      }
      // 减少所有触发冷却
      for (const [key, cd] of inst.cooldowns.entries()) {
        if (cd > 0) {
          inst.cooldowns.set(key, cd - 1)
        }
      }
    }
    // 移除过期的Buff
    this.removeExpiredBuffs(participantId)
  }

  private removeExpiredBuffs(participantId: string): void {
    const instances = this.activeBuffs.get(participantId) || []
    const remaining = instances.filter(inst => inst.remainingDuration !== 0)
    const removed = instances.filter(inst => inst.remainingDuration === 0)

    // 清理移除的Buff的监听器和修饰符
    for (const inst of removed) {
      for (const listenerId of inst.listenerIds) {
        // 需要知道原来注册的phase，这里简化：实际应存储 phase+id 映射
        // 可通过遍历所有phase尝试off，或改进数据结构
      }
      // 移除属性修饰符
      const stack = this.getModifierStack(participantId)
      if (stack) {
        stack.removeModifiersBySource(inst.buffId) // 需要ModifierStack支持
      }
    }

    this.activeBuffs.set(participantId, remaining)
  }

  // ... 其他必要方法
}
```

### 四、战斗流程中的事件触发

在战斗核心逻辑（如 `BattleEngine`）的各个阶段，调用 `eventBus.emit` 分发事件。

```typescript
// core/BattleEngine.ts (示例)

class BattleEngine {
  private eventBus: EventBus

  constructor() {
    this.eventBus = new EventBus()
  }

  // 攻击处理
  private processAttack(attacker: BattleParticipantImpl, defender: BattleParticipantImpl, skill: SkillConfig) {
    // 攻击前事件
    this.eventBus.emit('ON_ATTACK_BEFORE', {
      battleId: this.battleId,
      sourceId: attacker.id,
      targetId: defender.id,
      skillId: skill.id,
    })

    // 计算伤害
    const damage = this.calculateDamage(attacker, defender, skill)

    // 命中事件
    this.eventBus.emit('ON_ATTACK_HIT', {
      battleId: this.battleId,
      sourceId: attacker.id,
      targetId: defender.id,
      skillId: skill.id,
      value: damage,
    })

    // 应用伤害
    defender.takeDamage(damage)

    // 攻击后事件
    this.eventBus.emit('ON_ATTACK_AFTER', {
      battleId: this.battleId,
      sourceId: attacker.id,
      targetId: defender.id,
      skillId: skill.id,
      value: damage,
    })

    // 如果目标死亡，触发击杀/死亡事件
    if (!defender.isAlive()) {
      this.eventBus.emit('ON_KILL', {
        battleId: this.battleId,
        sourceId: attacker.id,
        targetId: defender.id,
      })
      this.eventBus.emit('ON_DEATH', {
        battleId: this.battleId,
        sourceId: defender.id,
        targetId: attacker.id,
      })
    }
  }

  // 回合开始
  private startTurn(participant: BattleParticipantImpl) {
    this.eventBus.emit('ON_TURN_START', {
      battleId: this.battleId,
      sourceId: participant.id,
    })
    // ... 执行行动
  }

  // 回合结束
  private endTurn(participant: BattleParticipantImpl) {
    this.eventBus.emit('ON_TURN_END', {
      battleId: this.battleId,
      sourceId: participant.id,
    })
    // 通知BuffSystem减少持续时间和冷却
    this.buffSystem.onTurnEnd(participant.id)
  }
}
```

### 六、方案优势总结

| 维度 | 说明 |
| :--- | :--- |
| **职责分离** | 属性计算只关心数值，事件系统只关心时机，两者通过BuffSystem桥接。 |
| **扩展性** | 新增触发阶段或行为脚本非常容易，只需在枚举和脚本处理中添加分支。 |
| **性能** | 事件监听采用发布订阅模式，无事件时不产生开销。属性计算仍保持高效。 |
| **可追溯性** | 属性修饰符来源依然清晰；触发行为可通过日志追踪。 |
| **易测试** | 事件总线、BuffSystem、属性引擎均可独立单元测试。 |

这个方案为你的战斗系统提供了一个坚实且灵活的基础，后续可以轻松接入行为树或更复杂的技能脚本系统。
