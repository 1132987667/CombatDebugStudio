## 一、数据流全链路

```
configs/*.json ──种子导入──▶ IndexedDB (DB_VERSION=6)
                                │
                    ┌───────────┴───────────┐
                    ▼                       ▼
          GameDataApi               BattleDataLoader
        (Promise 只读)            (启动预载 → IdbDataSource)
                    │                       │
                    ▼                       ▼
          GameDataProcessor ◀── IDataSource 抽象
           (同步出口)              ConfigDataSource 兜底
                    │
        ┌───────────┼───────────────┐
        ▼           ▼               ▼
   DataTable   EntityDetail    EntityDrawer
   (列表列)     Panel(详情)     (编辑表单)
        │           │               │
        └─────┬─────┘               │
              ▼                     ▼
        refNames.ts            FieldEditor
     (id→中文翻译)          (schema 驱动)
              │
              ▼
    fengshenStore.refIndex
   (全表引用字典，写后失效重建)
```

**写路径**：`EntityDrawer → FengshenDataService.save → DataIntegrityService.validate → IndexedDB.set → dataVersion++ → onDataChanged → BattleDataLoader.reload（非战斗中）+ refIndex 失效重建`

### A-1. 测试先行

**文件**：`src/domain/fengshen/__tests__/refNames.test.ts`

```typescript
import { describe, it, expect } from 'vitest'
import { resolveRefName, resolveRefNames, buildNameIndex } from '../refNames'

describe('refNames', () => {
  describe('buildNameIndex', () => {
    it('从单表文档数组构建 id→name 字典', () => {
      const docs = [
        { id: 'actor_001', name: '灵猴' },
        { id: 'actor_002', name: '石灵' },
      ]
      const index = buildNameIndex(docs)
      expect(index['actor_001']).toBe('灵猴')
      expect(index['actor_002']).toBe('石灵')
    })

    it('缺少 name 字段的文档以 id 自身作为回退值', () => {
      const docs = [{ id: 'actor_x' }]
      const index = buildNameIndex(docs)
      expect(index['actor_x']).toBe('actor_x')
    })

    it('空数组返回空字典', () => {
      expect(buildNameIndex([])).toEqual({})
    })
  })

  describe('resolveRefName', () => {
    const index: Record<string, string> = {
      actor_001: '灵猴',
      enemy_s1_1_a: '桃妖',
      skill_001: '花粉迷雾',
    }

    it('命中时返回中文名', () => {
      expect(resolveRefName('actor_001', index)).toBe('灵猴')
    })

    it('未命中时回退返回原始 id', () => {
      expect(resolveRefName('unknown_id', index)).toBe('unknown_id')
    })

    it('空字符串回退返回空字符串', () => {
      expect(resolveRefName('', index)).toBe('')
    })
  })

  describe('resolveRefNames', () => {
    const index: Record<string, string> = {
      actor_001: '灵猴',
      enemy_s1_1_a: '桃妖',
      skill_001: '花粉迷雾',
    }

    it('批量翻译保持顺序，逐项回退', () => {
      const result = resolveRefNames(['actor_001', 'missing', 'skill_001'], index)
      expect(result).toEqual(['灵猴', 'missing', '花粉迷雾'])
    })

    it('空数组返回空数组', () => {
      expect(resolveRefNames([], index)).toEqual([])
    })
  })

  describe('跨表合并（buildMergedNameIndex）', () => {
    it('多表文档合并为单一字典，跨表 id 均可命中', () => {
      // 模拟 REFERENCE_RULES 中 roles[].roleId → ['actors','enemies'] 的场景
      const actors = [
        { id: 'yaotu_fire', name: '火系护法' },
        { id: 'yaotu_gold', name: '金系护法' },
      ]
      const enemies = [
        { id: 'enemy_s1_1_a', name: '桃妖' },
        { id: 'boss_major_huayaowang', name: '花妖王·千瓣' },
      ]

      // 合并逻辑：遍历所有目标表，逐表 buildNameIndex 后 Object.assign
      const merged: Record<string, string> = {}
      for (const docs of [actors, enemies]) {
        Object.assign(merged, buildNameIndex(docs))
      }

      expect(merged['yaotu_fire']).toBe('火系护法')
      expect(merged['enemy_s1_1_a']).toBe('桃妖')
      expect(merged['boss_major_huayaowang']).toBe('花妖王·千瓣')

      // resolveRefNames 跨表命中
      const ids = ['yaotu_fire', 'enemy_s1_1_a', 'boss_major_huayaowang']
      expect(resolveRefNames(ids, merged)).toEqual(['火系护法', '桃妖', '花妖王·千瓣'])
    })

    it('后表同名 id 覆盖前表（后者胜出）', () => {
      const tableA = [{ id: 'dup', name: 'A表名' }]
      const tableB = [{ id: 'dup', name: 'B表名' }]
      const merged: Record<string, string> = {}
      Object.assign(merged, buildNameIndex(tableA))
      Object.assign(merged, buildNameIndex(tableB))
      expect(merged['dup']).toBe('B表名')
    })
  })
})
```

```typescript
// ─── refIndex 重建（跨表合并）───
async function ensureRefIndex(): Promise<void> {
  if (refIndexLoaded) return
  const merged: Record<string, string> = {}

  // 遍历 REFERENCE_RULES 中所有 targetTable（支持 string | string[]）
  const targetTables = new Set<string>()
  for (const rule of REFERENCE_RULES) {
    const tables = Array.isArray(rule.targetTable) ? rule.targetTable : [rule.targetTable]
    for (const t of tables) targetTables.add(t)
  }

  // 逐表拉取 → buildNameIndex → 合并
  for (const table of targetTables) {
    try {
      const docs = await GameDataApi.listByTable(table)
      Object.assign(merged, buildNameIndex(docs))
    } catch {
      // 表不存在或为空时静默跳过，不阻断其他表
    }
  }

  refIndex.value = merged
  refIndexLoaded = true
}
```

**关键改动点**：若原实现仅遍历当前 `activeTable`，改为遍历 `REFERENCE_RULES` 中声明的全部 `targetTable`（去重后）。

```typescript
/** 字段级 diff 条目（与 FengshenDataService 写入的 FieldDiff 对齐） */
export interface FieldDiff {
  field: string
  oldValue: unknown
  newValue: unknown
}

/**
 * 解析操作日志的 detail 字段（JSON 序列化的 FieldDiff[]）。
 * 畸形输入兜底返回空数组，不抛异常。
 */
export function parseLogDetail(detail: string | undefined | null): FieldDiff[] {
  if (!detail) return []
  try {
    const parsed = JSON.parse(detail)
    if (!Array.isArray(parsed)) return []
    return parsed as FieldDiff[]
  } catch {
    return []
  }
}

/**
 * 将 diff 值格式化为可读字符串。
 * - null/undefined → 「空」
 * - boolean → 是/否
 * - 对象/数组 → JSON 截断（≤60 字符）
 * - 其他 → String()
 */
export function formatDiffValue(v: unknown): string {
  if (v === null || v === undefined) return '空'
  if (typeof v === 'boolean') return v ? '是' : '否'
  if (typeof v === 'object') {
    const json = JSON.stringify(v)
    return json.length > 60 ? json.slice(0, 60) + '…' : json
  }
  return String(v)
}
```

### D-2. 实现：纯函数

**新建文件**：`src/domain/fengshen/validation.ts`

```typescript
/**
 * 引用字段即时校验（编辑态 onBlur 调用）。
 * 返回错误信息字符串；通过校验返回 null。
 *
 * 支持单值（"actor_001"）与逗号分隔多值（"actor_001,actor_002"）。
 * 空值/纯空白视为可选字段，不报错。
 */
export function validateRefField(
  value: string,
  index: Record<string, string>,
): string | null {
  const trimmed = value.trim()
  if (!trimmed) return null

  const ids = trimmed.split(',').map((s) => s.trim()).filter(Boolean)
  for (const id of ids) {
    if (!(id in index)) {
      return `「${id}」在引用表中未找到`
    }
  }
  return null
}
```
