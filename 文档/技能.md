export class BattleParticipantImpl implements BattleParticipant {
  private attributes = new Map<ATTRIBUTE_CODE, AttributeValue>()
  private modifierProvider: IModifierProvider

  constructor(data: ParticipantInitData, modifierProvider: IModifierProvider) {
    this.modifierProvider = modifierProvider
    // ... 初始化基础字段
    this.initAttribute(ATTRIBUTES.MAX_HP, data.maxHealth)
    this.initAttribute(ATTRIBUTES.HP, data.currentHealth ?? data.maxHealth)
    // ... 其他属性
    this.markAllDirty()
  }

  private initAttribute(
    type: ATTRIBUTE_CODE,
    base: number,
    isPercentage = false,
  ) {
    this.attributes.set(type, {
      value: base,
      base,
      modifiers: [],
      isPercentage,
      dirty: true,
    })
  }

  markDirty(type: ATTRIBUTE_CODE) {
    this.attributes.get(type)!.dirty = true
  }

  markAllDirty() {
    this.attributes.forEach((attr) => (attr.dirty = true))
  }

  private recalcAttribute(type: ATTRIBUTE_CODE) {
    const attr = this.attributes.get(type)
    if (!attr || !attr.dirty) return
    const modifiers =
      this.modifierProvider.getModifierStack(this.id)?.getModifiers(type) ?? []
    // ... 多乘区计算（参考文档）
    attr.value = finalValue
    attr.modifiers = details
    attr.dirty = false
  }

  getAttributeValue(type: ATTRIBUTE_CODE): AttributeValue | undefined {
    const attr = this.attributes.get(type)
    if (attr?.dirty) this.recalcAttribute(type)
    return attr
  }

  getAttribute(type: ATTRIBUTE_CODE): number {
    return this.getAttributeValue(type)?.value ?? 0
  }

  recalculateAll() {
    this.attributes.forEach((_, type) => this.recalcAttribute(type))
  }

  // 原有的 currentHealth getter/setter 需要改为通过 attributes 实现
  get currentHealth(): number {
    return this.getAttribute(ATTRIBUTES.HP)
  }
  set currentHealth(value: number) {
    const attr = this.attributes.get(ATTRIBUTES.HP)
    if (attr) attr.value = Math.min(Math.max(0, value), this.maxHealth)
  }
  // 类似处理 maxHealth, currentEnergy 等
}

## 三、迁移注意事项

1. **属性访问替换**：使用正则或手动将所有 `.currentHp` 替换为 `.getAttributeValue('HP').value`，但注意 `.value` 是 `AttributeValue` 对象的属性，不要混淆。
2. **响应式优化**：在 Vue 组件中，务必使用 `computed` 缓存 `getAttributeValue` 的结果，避免在模板中频繁调用方法导致性能下降。
3. **调试模式**：生产环境关闭 `breakdown` 生成，通过 `isDebugMode()` 控制。
4. **序列化**：战斗回放保存时，需要将 `AttributeValue` 对象简化为只存 `value` 和 `base`，加载时重建。
5. **测试**：确保所有 UI 显示、伤害计算、Buff 效果正常，无性能回退。
